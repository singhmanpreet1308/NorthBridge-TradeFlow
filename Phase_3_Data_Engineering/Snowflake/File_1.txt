//Verify the table
DESCRIBE TABLE NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED;



USE ROLE ACCOUNTADMIN;

-- Allow access to the CURATED schema
GRANT USAGE
ON SCHEMA NORTHBRIDGE_TRADEFLOW.CURATED
TO ROLE NORTHBRIDGE_INGEST_ROLE;

-- Allow inserting and reading validated trades
GRANT INSERT, SELECT
ON TABLE NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
TO ROLE NORTHBRIDGE_INGEST_ROLE;


// Verify Snowflake Ingestion
SELECT
    VALIDATION_RUN_ID,
    VALIDATION_STATUS,
    COUNT(*) AS TOTAL_TRADES
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
GROUP BY
    VALIDATION_RUN_ID,
    VALIDATION_STATUS;


// Verify the record count
SELECT COUNT(*) AS TOTAL_TRADES
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
WHERE VALIDATION_RUN_ID = 'VAL-E6532CDFF204';


SELECT
    COUNT(*) AS TOTAL_RECORDS,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADE_IDS,
    COUNT_IF(VALIDATION_STATUS = 'PASS') AS PASSED_RECORDS,
    COUNT_IF(VALIDATION_STATUS <> 'PASS'
             OR VALIDATION_STATUS IS NULL) AS NON_PASSED_RECORDS,
    COUNT_IF(TRADE_ID IS NULL) AS NULL_TRADE_IDS
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
WHERE VALIDATION_RUN_ID = 'VAL-E6532CDFF204';

//Verify the Final Snowflake Record Count
SELECT
    COUNT(*) AS TOTAL_TRADES,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED;

// Verify Validation Status in Snowflake
SELECT
    VALIDATION_STATUS,
    COUNT(*) AS TRADE_COUNT
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
GROUP BY VALIDATION_STATUS
ORDER BY VALIDATION_STATUS;
