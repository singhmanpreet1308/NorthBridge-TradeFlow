// Check Existing Permissions Verification
SHOW GRANTS TO ROLE NORTHBRIDGE_INGEST_ROLE;

// Create the dbt Development Schema
USE ROLE ACCOUNTADMIN;

/*
CREATE SCHEMA IF NOT EXISTS
NORTHBRIDGE_TRADEFLOW.DBT_DEV;
*/
GRANT USAGE
ON SCHEMA NORTHBRIDGE_TRADEFLOW.DBT_DEV
TO ROLE NORTHBRIDGE_INGEST_ROLE;

GRANT CREATE TABLE, CREATE VIEW
ON SCHEMA NORTHBRIDGE_TRADEFLOW.DBT_DEV


// Verify the 49 Trades in Snowflake
USE ROLE ACCOUNTADMIN;

SELECT
    COUNT(*) AS TOTAL_TRADES,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.STG_TRADES_VALIDATED;

//Inspect the Staging Columns
DESC VIEW NORTHBRIDGE_TRADEFLOW.DBT_DEV.STG_TRADES_VALIDATED;



// Retrieve All 26 Columns
  
DESC VIEW NORTHBRIDGE_TRADEFLOW.DBT_DEV.STG_TRADES_VALIDATED;

SELECT
    "name" AS COLUMN_NAME,
    "type" AS DATA_TYPE
FROM TABLE(RESULT_SCAN(LAST_QUERY_ID()));


// Verify the Trade Enrichment Model
USE ROLE ACCOUNTADMIN;
SELECT
    TRADE_ID,
    EFFECTIVE_DATE,
    MATURITY_DATE,
    SWAP_TENOR_YEARS,
    REMAINING_MATURITY_YEARS
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_ENRICHED
LIMIT 10;

// Verify Automatic SELECT Grants
SHOW GRANTS ON VIEW
NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_ENRICHED;

// Verify Counterparty Exposure and Reconcile Trade Counts
USE ROLE ACCOUNTADMIN;

SELECT
    COUNTERPARTY_ID,
    CURRENCY,
    TOTAL_TRADES,
    GROSS_NOTIONAL,
    AVG_TRADE_NOTIONAL
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_COUNTERPARTY_EXPOSURE
ORDER BY GROSS_NOTIONAL DESC;


SELECT
    SUM(TOTAL_TRADES) AS MART_TRADE_COUNT,
    49 AS EXPECTED_TRADE_COUNT,
    SUM(TOTAL_TRADES) - 49 AS DIFFERENCE
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_COUNTERPARTY_EXPOSURE;

// Verify the Portfolio Summary

USE ROLE ACCOUNTADMIN;

SELECT
    CURRENCY,
    TOTAL_TRADES,
    GROSS_NOTIONAL,
    AVG_TRADE_NOTIONAL,
    AVG_SWAP_TENOR_YEARS,
    MATURED_TRADES,
    NOT_YET_MATURED_TRADES
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_TRADE_PORTFOLIO_SUMMARY;

// Verify the trade counts

SELECT
    SUM(TOTAL_TRADES) AS TOTAL_TRADES,
    SUM(MATURED_TRADES) AS MATURED_TRADES,
    SUM(NOT_YET_MATURED_TRADES) AS NOT_YET_MATURED_TRADES,
    SUM(MATURED_TRADES + NOT_YET_MATURED_TRADES)
        AS CLASSIFIED_TRADES
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_TRADE_PORTFOLIO_SUMMARY;

// Verify Current Trade Selection
USE ROLE ACCOUNTADMIN;

SELECT
    COUNT(*) AS TOTAL_TRADES,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES,
    COUNT(*) - COUNT(DISTINCT TRADE_ID) AS DUPLICATES
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES;

// Top 10
SELECT
    TRADE_ID,
    TRADE_VERSION,
    VALIDATED_AT_UTC
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
ORDER BY TRADE_ID
LIMIT 10;

// Verify Trade-Version History

USE ROLE ACCOUNTADMIN;

SELECT
    COUNT(*) AS TOTAL_RECORDS,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES,
    COUNT_IF(IS_LATEST_VERSION = TRUE) AS LATEST_VERSIONS,
    MAX(VERSION_COUNT) AS MAX_VERSION_COUNT
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_VERSION_HISTORY;

// Check individual trade versions

SELECT
    TRADE_ID,
    TRADE_VERSION,
    VERSION_RANK,
    VERSION_COUNT,
    IS_LATEST_VERSION
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_VERSION_HISTORY
ORDER BY TRADE_ID
LIMIT 10;


// Verify the Validation Summary in Snowflake
USE ROLE ACCOUNTADMIN;

SELECT
    BATCH_ID,
    TOTAL_TRADES,
    CURATED_RECORDS,
    QUARANTINE_RECORDS,
    ACCEPTANCE_RATE_PCT,
    QUARANTINE_RATE_PCT,
    RECONCILIATION_STATUS
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_TRADE_VALIDATION_SUMMARY;
