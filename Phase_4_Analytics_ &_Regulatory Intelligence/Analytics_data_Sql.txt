// Prepare the analytics data
SHOW WAREHOUSES;


USE WAREHOUSE NORTHBRIDGE_WH;

USE DATABASE NORTHBRIDGE_TRADEFLOW;

SELECT
    TABLE_SCHEMA,
    TABLE_NAME,
    TABLE_TYPE
FROM NORTHBRIDGE_TRADEFLOW.INFORMATION_SCHEMA.TABLES
WHERE TABLE_SCHEMA <> 'INFORMATION_SCHEMA'
ORDER BY TABLE_SCHEMA, TABLE_NAME;

// Inspect our analytics tables
SELECT
    TABLE_NAME,
    COLUMN_NAME,
    DATA_TYPE
FROM NORTHBRIDGE_TRADEFLOW.INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'DBT_DEV'
  AND TABLE_NAME IN (
      'INT_CURRENT_TRADES',
      'MART_TRADE_VALIDATION_SUMMARY'
  )
ORDER BY TABLE_NAME, ORDINAL_POSITION;

// Calculate KPI: Total Trade Volume
USE WAREHOUSE NORTHBRIDGE_WH;
USE DATABASE NORTHBRIDGE_TRADEFLOW;

SELECT
    COUNT(*) AS TOTAL_CURRENT_RECORDS,
    COUNT(DISTINCT TRADE_ID) AS TOTAL_UNIQUE_TRADES,
    COUNT(*) - COUNT(DISTINCT TRADE_ID) AS DUPLICATE_COUNT,
    SUM(NOTIONAL_AMOUNT) AS TOTAL_NOTIONAL_GBP
FROM DBT_DEV.INT_CURRENT_TRADES
WHERE CURRENCY = 'GBP';

//Inspect Validation Status
SELECT
    VALIDATION_STATUS,
    COUNT(*) AS TRADE_COUNT
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
GROUP BY VALIDATION_STATUS
ORDER BY TRADE_COUNT DESC;

// Locate T-1001 in the source data
SELECT
    'STAGING.TRADES_RAW' AS SOURCE_TABLE,
    COUNT(*) AS MATCHING_RECORDS
FROM NORTHBRIDGE_TRADEFLOW.STAGING.TRADES_RAW
WHERE TRIM(UPPER(TRADE_ID)) = 'T-1001'

UNION ALL

SELECT
    'CURATED.TRADES_VALIDATED' AS SOURCE_TABLE,
    COUNT(*) AS MATCHING_RECORDS
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
WHERE TRIM(UPPER(TRADE_ID)) = 'T-1001'

UNION ALL

SELECT
    'DBT_DEV.INT_CURRENT_TRADES' AS SOURCE_TABLE,
    COUNT(*) AS MATCHING_RECORDS
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
WHERE TRIM(UPPER(TRADE_ID)) = 'T-1001';


// Record Count Verification
SELECT
    'STAGING.TRADES_RAW' AS TABLE_NAME,
    COUNT(*) AS RECORD_COUNT
FROM NORTHBRIDGE_TRADEFLOW.STAGING.TRADES_RAW

UNION ALL

SELECT
    'CURATED.TRADES_VALIDATED',
    COUNT(*)
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED

UNION ALL

SELECT
    'DBT_DEV.INT_CURRENT_TRADES',
    COUNT(*)
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES

UNION ALL

SELECT
    'DBT_DEV.INT_TRADE_VERSION_HISTORY',
    COUNT(*)
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_VERSION_HISTORY;


// Inspect the dbt source relationship
SELECT GET_DDL(
    'VIEW',
    'NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_TRADE_VERSION_HISTORY'
) AS VIEW_DEFINITION;


// Check Trade IDs in the curated dataset
SELECT
    TRADE_ID,
    TRADE_VERSION,
    NOTIONAL_AMOUNT,
    TRADE_STATUS
FROM NORTHBRIDGE_TRADEFLOW.CURATED.TRADES_VALIDATED
ORDER BY TRADE_ID
LIMIT 15;

// Check the total & Unique records
SELECT
    COUNT(*) AS TOTAL_RECORDS,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES,
    COUNT_IF(
        TRADE_ID = 'T-1001'
        AND TRADE_VERSION = 2
    ) AS T1001_V2_COUNT,
    COUNT_IF(
        VALIDATION_RUN_ID = 'VAL-90BA7E7183AC'
    ) AS EXISTING_VALIDATION_RUN_RECORDS
FROM CURATED.TRADES_VALIDATED;

// Verify the ingestion amendment result
SELECT
    COUNT(*) AS TOTAL_RECORDS,
    COUNT(DISTINCT TRADE_ID) AS UNIQUE_TRADES,
    COUNT_IF(
        TRADE_ID = 'T-1001'
        AND TRADE_VERSION = 2
    ) AS T1001_V2_COUNT,
    COUNT(*) - COUNT(DISTINCT TRADE_ID, TRADE_VERSION)
        AS DUPLICATE_VERSION_COUNT
FROM CURATED.TRADES_VALIDATED;

// Verify Updated Analytics Views
SELECT
    'CURRENT_TRADES' AS DATASET,
    COUNT(*) AS RECORD_COUNT
FROM DBT_DEV.INT_CURRENT_TRADES

UNION ALL

SELECT
    'TRADE_VERSION_HISTORY',
    COUNT(*)
FROM DBT_DEV.INT_TRADE_VERSION_HISTORY

UNION ALL

SELECT
    'TRADE_ENRICHED',
    COUNT(*)
FROM DBT_DEV.INT_TRADE_ENRICHED;

// Verify 'T-1001' in the Analytics Layer
SELECT
    TRADE_ID,
    TRADE_VERSION,
    NOTIONAL_AMOUNT,
    TRADE_STATUS,
    AMENDMENT_REASON,
    VALIDATION_STATUS,
    VALIDATION_RUN_ID
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
WHERE TRADE_ID = 'T-1001';

// Inspect Current Trade Status Distribution
SELECT
    TRADE_STATUS,
    VALIDATION_STATUS,
    COUNT(*) AS TRADE_COUNT
FROM DBT_DEV.INT_CURRENT_TRADES
GROUP BY
    TRADE_STATUS,
    VALIDATION_STATUS
ORDER BY TRADE_COUNT DESC;

// Current Trade KPI Verification
SELECT
    COUNT(DISTINCT TRADE_ID) AS TOTAL_TRADES,

    SUM(
        CASE
            WHEN CURRENCY = 'GBP'
            THEN NOTIONAL_AMOUNT
            ELSE 0
        END
    ) AS TOTAL_NOTIONAL_GBP,

    COUNT_IF(TRADE_STATUS = 'AMENDED')
        AS AMENDED_TRADES,

    COUNT_IF(VALIDATION_STATUS = 'PASS')
        AS VALIDATED_TRADES,

    COUNT_IF(VALIDATION_STATUS = 'BLOCK')
        AS BLOCKED_CURRENT_TRADES,

    ROUND(
        100.0 * COUNT_IF(VALIDATION_STATUS = 'PASS')
        / NULLIF(COUNT(*), 0),
        2
    ) AS CURRENT_VALIDATION_PASS_RATE_PCT,

    ROUND(
        100.0 * COUNT_IF(VALIDATION_STATUS = 'BLOCK')
        / NULLIF(COUNT(*), 0),
        2
    ) AS CURRENT_VALIDATION_FAILURE_RATE_PCT

FROM DBT_DEV.INT_CURRENT_TRADES;


SELECT
    BATCH_ID,
    VALIDATION_RUN_ID,
    TOTAL_TRADES,
    CURATED_RECORDS,
    QUARANTINE_RECORDS,
    ROUND(
        100.0 * QUARANTINE_RECORDS
        / NULLIF(TOTAL_TRADES, 0),
        2
    ) AS QUARANTINE_RATE_PCT,
    RECONCILIATION_STATUS
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.MART_TRADE_VALIDATION_SUMMARY
ORDER BY VALIDATED_AT_UTC DESC;

// Check Available Processing Evidence
SELECT
    TABLE_SCHEMA,
    TABLE_NAME,
    TABLE_TYPE
FROM NORTHBRIDGE_TRADEFLOW.INFORMATION_SCHEMA.TABLES
WHERE TABLE_SCHEMA <> 'INFORMATION_SCHEMA'
  AND (
       TABLE_NAME ILIKE '%EVENT%'
    OR TABLE_NAME ILIKE '%REPORT%'
    OR TABLE_NAME ILIKE '%SUBMISSION%'
    OR TABLE_NAME ILIKE '%EXCEPTION%'
    OR TABLE_NAME ILIKE '%CONFIRMATION%'
  )
ORDER BY TABLE_SCHEMA, TABLE_NAME;

// Check Processing Timestamps
SELECT
    TRADE_ID,
    TRADE_VERSION,
    CAPTURE_TIMESTAMP,
    VALIDATED_AT_UTC,
    DATEDIFF(
        'second',
        CAPTURE_TIMESTAMP,
        VALIDATED_AT_UTC
    ) AS PROCESSING_SECONDS
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
ORDER BY TRADE_ID
LIMIT 10;
