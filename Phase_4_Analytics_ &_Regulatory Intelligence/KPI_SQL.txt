
-- NorthBridge TradeFlow
-- Phase 4: Analytics and Regulatory Intelligence
-- KPI reconciliation queries

USE WAREHOUSE NORTHBRIDGE_WH;
USE DATABASE NORTHBRIDGE_TRADEFLOW;

-- 1. Current trade KPIs
-- Grain: one latest accepted version per trade

SELECT
    COUNT(DISTINCT TRADE_ID) AS TOTAL_TRADES,

    SUM(
        CASE
            WHEN CURRENCY = 'GBP'
            THEN NOTIONAL_AMOUNT
            ELSE 0
        END
    ) AS TOTAL_NOTIONAL_GBP,

    COUNT_IF(TRADE_STATUS = 'AMENDED')
        AS AMENDED_TRADES,

    COUNT_IF(VALIDATION_STATUS = 'PASS')
        AS VALIDATED_TRADES,

    COUNT_IF(VALIDATION_STATUS = 'BLOCK')
        AS BLOCKED_CURRENT_TRADES,

    ROUND(
        100.0 * COUNT_IF(VALIDATION_STATUS = 'PASS')
        / NULLIF(COUNT(*), 0),
        2
    ) AS CURRENT_PASS_RATE_PCT,

    ROUND(
        100.0 * COUNT_IF(VALIDATION_STATUS = 'BLOCK')
        / NULLIF(COUNT(*), 0),
        2
    ) AS CURRENT_FAILURE_RATE_PCT

FROM DBT_DEV.INT_CURRENT_TRADES;


-- 2. Initial validation batch
-- Grain: one validation batch

SELECT
    BATCH_ID,
    VALIDATION_RUN_ID,
    TOTAL_TRADES,
    CURATED_RECORDS,
    QUARANTINE_RECORDS,

    ROUND(
        100.0 * QUARANTINE_RECORDS
        / NULLIF(TOTAL_TRADES, 0),
        2
    ) AS QUARANTINE_RATE_PCT,

    RECONCILIATION_STATUS

FROM DBT_DEV.MART_TRADE_VALIDATION_SUMMARY
WHERE BATCH_ID = 'BATCH-60FD5666D0A1FBAD';


-- 3. Trade status distribution

SELECT
    TRADE_STATUS,
    COUNT(DISTINCT TRADE_ID) AS TRADE_COUNT
FROM DBT_DEV.INT_CURRENT_TRADES
GROUP BY TRADE_STATUS
ORDER BY TRADE_COUNT DESC;


-- 4. GBP notional by counterparty

SELECT
    COUNTERPARTY_ID,
    SUM(NOTIONAL_AMOUNT) AS TOTAL_NOTIONAL_GBP
FROM DBT_DEV.INT_CURRENT_TRADES
WHERE CURRENCY = 'GBP'
GROUP BY COUNTERPARTY_ID
ORDER BY TOTAL_NOTIONAL_GBP DESC;


-- 5. T-1001 corrected trade

SELECT
    TRADE_ID,
    TRADE_VERSION,
    NOTIONAL_AMOUNT,
    TRADE_STATUS,
    VALIDATION_STATUS,
    AMENDMENT_REASON,
    VALIDATION_RUN_ID
FROM DBT_DEV.INT_CURRENT_TRADES
WHERE TRADE_ID = 'T-1001';


-- 6. Duplicate version verification

SELECT
    TRADE_ID,
    TRADE_VERSION,
    COUNT(*) AS RECORD_COUNT
FROM CURATED.TRADES_VALIDATED
GROUP BY TRADE_ID, TRADE_VERSION
HAVING COUNT(*) > 1;
