// Verify
 SELECT
    COUNT(*) AS TOTAL_SUBMISSIONS,
    COUNT_IF(REPORTING_STATUS = 'ACCEPTED') AS ACCEPTED,
    COUNT_IF(REPORTING_STATUS = 'REJECTED') AS REJECTED,
    ROUND(
        100.0 * COUNT_IF(REPORTING_STATUS = 'REJECTED')
        / NULLIF(COUNT(*), 0), 2
    ) AS SIMULATED_REJECT_RATE_PCT
FROM CURATED.SIMULATED_REPORTING_OUTCOMES;

// Check available lifecycle evidence
SELECT
    TRADE_STATUS,
    COUNT(*) AS TRADE_COUNT,
    COUNT_IF(AMENDMENT_REASON IS NOT NULL) AS AMENDED_WITH_REASON
FROM NORTHBRIDGE_TRADEFLOW.DBT_DEV.INT_CURRENT_TRADES
GROUP BY TRADE_STATUS
ORDER BY TRADE_COUNT DESC;
